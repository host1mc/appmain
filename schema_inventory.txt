================================================================================
 dc_hostfnal — FULL TABLE INVENTORY (two stores)
================================================================================
 ATP / Oracle  :  9 tables — identity, PII, anything transactional. FATAL if lost.
 HeatWave/MySQL:  6 tables — bots + public/ops data. Bots degrade gracefully,
                  the rest is losable.

 Post-consolidation state: the Oracle side no longer owns bots, hosting or
 panel tables. Removed outright: bots (Oracle copy — moved to HeatWave),
 hosting_servers, hosting_backups, fingerprint_history (merged into
 fingerprints), panel_users, panel_servers, panel_activity. Removed from
 users: discord_bot_token, webhook_token (live on the HeatWave bots rows),
 inactive_warned_at (renewed weekly, marker lives in HeatWave app_config).

 Growth flags:  tiny | bounded | UNBOUNDED(*space hog)
================================================================================

# ATP / ORACLE  (database.py -> _oracle_conn)

-- users : accounts (auth, bans, slots, trial)                         [tiny]
-- "uid" is the one identifier: random 10-char string, join key for every
-- user-linked table (Oracle and HeatWave). Quoted because UID is an Oracle
-- pseudocolumn. No separate id/user_id pair.
CREATE TABLE users (
    "uid"                    VARCHAR2(10)   PRIMARY KEY,
    username                 VARCHAR2(255)  UNIQUE NOT NULL,
    username_lookup_hash     VARCHAR2(64)   UNIQUE,
    username_ci_lookup_hash  VARCHAR2(64),
    email                    VARCHAR2(255),
    email_lookup_hash        VARCHAR2(64)   UNIQUE,
    password                 VARCHAR2(255),
    display_name             VARCHAR2(255),
    slots                    VARCHAR2(10)   DEFAULT '1',   -- bot-slot grant
    container_slots          VARCHAR2(10)   DEFAULT '1',   -- hosting grant
    email_verified           VARCHAR2(5)    DEFAULT '0',
    github_verified          VARCHAR2(5)    DEFAULT '0',
    account_type             VARCHAR2(20)   DEFAULT 'trial',
    trial_expires_at         VARCHAR2(50),
    is_active                NUMBER         DEFAULT 1,
    created_at               VARCHAR2(50)   NOT NULL,
    last_login               VARCHAR2(50),
    ads_disabled             NUMBER         DEFAULT 0,
    is_banned                NUMBER         DEFAULT 0,
    banned_reason            VARCHAR2(2000),
    fingerprint_ip           VARCHAR2(500),   -- encrypted
    verified_at              VARCHAR2(50),
    last_active_at           VARCHAR2(50),
    bot_stopped_at           VARCHAR2(50),
    banned_at                VARCHAR2(50),
    banned_purged_at         VARCHAR2(50)
);

-- servers : ONE table for every container (hosting + panel merged)   [bounded]
-- Holds identity, intent and placement only. Runtime, version, memory and the
-- startup command are NOT columns — they live on the container itself (Docker
-- labels read by the node agent), hidden from the panel DB.
-- status is the desired-state flag (0 stopped / 1 running).
CREATE TABLE servers (
    id          VARCHAR2(36)  PRIMARY KEY,
    "uid"       VARCHAR2(10)  NOT NULL,
    name        VARCHAR2(255),
    status      NUMBER        DEFAULT 0 NOT NULL,
    node_id     NUMBER,                        -- placement; NULL = unplaced
    created_at  DATE          NOT NULL,
    FOREIGN KEY ("uid") REFERENCES users ("uid")
);
-- INDEX servers_uid ("uid")

-- fingerprints : device bindings, MERGED with the old fingerprint_history  [UNBOUNDED*]
-- Append-only: every sighting is a row; the newest row per uid IS the current
-- binding (the separate "current" table and history table are one). Reset
-- deletes the user's rows; housekeeping keeps the newest row per uid.
CREATE TABLE fingerprints (
    id                NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    "uid"             VARCHAR2(10)  NOT NULL,
    fingerprint_hash  VARCHAR2(500) NOT NULL,
    lookup_hash       VARCHAR2(500) NOT NULL,
    device_info_enc   CLOB,
    ip_address        VARCHAR2(255),
    ip_lookup_hash    VARCHAR2(64),
    created_at        VARCHAR2(50)  NOT NULL,
    FOREIGN KEY ("uid") REFERENCES users ("uid")
);
-- INDEX fp_uid ("uid"), fp_lookup (lookup_hash)

-- device_events : security / audit events                  [UNBOUNDED *space hog]
CREATE TABLE device_events (
    id                NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    "uid"             VARCHAR2(10),
    username          VARCHAR2(255),
    event_type        VARCHAR2(50)  NOT NULL,
    lookup_hash       VARCHAR2(500),
    fingerprint_enc   VARCHAR2(500),
    device_info_enc   CLOB,
    ip_address        VARCHAR2(45),
    blocked           NUMBER        DEFAULT 0,
    reviewed          NUMBER        DEFAULT 0,
    details           CLOB,
    created_at        VARCHAR2(50)  NOT NULL
);

-- settings : global key/value                                         [tiny]
CREATE TABLE settings (
    key    VARCHAR2(255) PRIMARY KEY,
    value  CLOB
);

-- sessions : login sessions                                           [bounded]
CREATE TABLE sessions (
    id           VARCHAR2(64)  PRIMARY KEY,
    "uid"        VARCHAR2(10),
    data         CLOB          NOT NULL,
    ip_address   VARCHAR2(45),
    user_agent   VARCHAR2(500),
    created_at   VARCHAR2(50)  NOT NULL,
    last_access  VARCHAR2(50)  NOT NULL,
    expires_at   VARCHAR2(50)  NOT NULL
);

-- user_ad_zone_overrides : per-user ad toggles                        [tiny]
CREATE TABLE user_ad_zone_overrides (
    id        NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    "uid"     VARCHAR2(10)  NOT NULL,
    zone_key  VARCHAR2(100) NOT NULL,
    enabled   NUMBER        DEFAULT 1,
    UNIQUE("uid", zone_key)
);

-- otp_codes : email OTP codes                                         [tiny]
CREATE TABLE otp_codes (
    id                 NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    email              VARCHAR2(255) NOT NULL,
    email_lookup_hash  VARCHAR2(64),
    code               VARCHAR2(255) NOT NULL,
    purpose            VARCHAR2(20)  DEFAULT 'register',
    expires_at         VARCHAR2(50)  NOT NULL,
    used               NUMBER        DEFAULT 0,
    attempts           NUMBER        DEFAULT 0,
    created_at         VARCHAR2(50)  NOT NULL
);

# HEATWAVE / MYSQL  (reviews_db.py -> _conn ; degrade-safe, None/[]/False on

-- bots : MC-status Discord bots, moved from Oracle                   [bounded]
-- Identified by (uid, slot_index) — deliberately NO synthetic id column.
-- uid is the link to the Oracle users table in the logical sense only (two
-- stores cannot share a constraint; same shape as reviews.uid). The Discord
-- token (token_enc) and webhook_url are Fernet-encrypted at rest; name,
-- server_ip, guild_id, channel_id, embed_json, ip_reply_json are too.
CREATE TABLE bots (
    uid              VARCHAR(10)   NOT NULL,
    slot_index       INT           NOT NULL DEFAULT 0,
    name             VARCHAR(500),
    server_ip        VARCHAR(500),
    server_port      INT           DEFAULT 25565,
    edition          VARCHAR(20)   DEFAULT 'java',
    token_enc        VARCHAR(1000),
    guild_id         VARCHAR(500),
    channel_id       VARCHAR(500),
    message_id       VARCHAR(100),
    embed_json       LONGTEXT,
    ip_reply_json    LONGTEXT,
    webhook_url      VARCHAR(1000),
    update_interval  INT           DEFAULT 60,
    running          TINYINT       DEFAULT 0,
    last_run         VARCHAR(50),
    last_status      LONGTEXT,
    last_error       VARCHAR(500),
    created_at       VARCHAR(50)   NOT NULL,
    updated_at       VARCHAR(50),
    PRIMARY KEY (uid, slot_index),
    KEY bots_running (running)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- reviews : public marketing reviews                                 [small]
CREATE TABLE reviews (
    id           INT AUTO_INCREMENT PRIMARY KEY,
    uid          VARCHAR(10)   NOT NULL,
    author_name  VARCHAR(100)  NOT NULL,
    rating       INT           NOT NULL,
    body         VARCHAR(4000) NOT NULL,
    embed_json   LONGTEXT,
    approved     TINYINT       NOT NULL DEFAULT 0,
    created_at   VARCHAR(50)   NOT NULL,
    KEY reviews_approved (approved)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- app_errors : error log (deduped via occurrences)                   [ops growth]
CREATE TABLE app_errors (
    id              INT AUTO_INCREMENT PRIMARY KEY,
    error_type      VARCHAR(255)  NOT NULL,
    message         TEXT          NOT NULL,
    stack_trace     LONGTEXT,
    module          VARCHAR(255),
    flag_reason     VARCHAR(255),
    flagged         TINYINT       NOT NULL DEFAULT 1,
    created_at      VARCHAR(50)   NOT NULL,
    error_category  VARCHAR(50)   DEFAULT 'system_error',   -- ALTER-added
    occurrences     INT           NOT NULL DEFAULT 1,       -- ALTER-added
    KEY errors_flagged (flagged)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- app_config : runtime config                                        [tiny]
-- Also holds the weekly inactivity marker per account as
-- "inactive_warned:<uid>" — the replacement for users.inactive_warned_at.
CREATE TABLE app_config (
    config_key    VARCHAR(100) PRIMARY KEY,
    config_value  TEXT         NOT NULL,
    updated_at    VARCHAR(50)  NOT NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- pending_container_deletions : deferred-delete queue                [tiny]
CREATE TABLE pending_container_deletions (
    server_id     VARCHAR(64) PRIMARY KEY,
    node_id       VARCHAR(64) NOT NULL DEFAULT '',
    node_name     VARCHAR(100) NOT NULL DEFAULT '',
    node_ip       VARCHAR(255) NOT NULL DEFAULT '',
    `purge`       TINYINT     NOT NULL DEFAULT 1,   -- backticked (reserved word)
    requested_at  VARCHAR(50) NOT NULL,
    KEY pcd_node (node_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- retired_nodes : decommissioned nodes                               [tiny]
CREATE TABLE retired_nodes (
    node_id     VARCHAR(64)   PRIMARY KEY,
    name        VARCHAR(100)  NOT NULL DEFAULT '',
    url         VARCHAR(255)  NOT NULL DEFAULT '',
    token_enc   VARCHAR(2000) NOT NULL DEFAULT '',
    retired_at  VARCHAR(50)   NOT NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

================================================================================
 NOTES ON THE MISMATCHES
================================================================================
 The MISMATCH rows are two DIFFERENT tables that happen to share a column NAME;
 none of them are a foreign-key pair enforced by the DB, so nothing breaks today.
 They are cosmetic/consistency drift EXCEPT:

  * node_id : NUMBER in servers/nodes vs VARCHAR(64) in the MySQL delete/retire
    queue. These DO refer to the same node identity across stores, so a lookup
    that carries a node id from one to the other must cast. The node-registry
    and deletion-queue code paths stringify consistently.

  * created_at : VARCHAR2(50) in most database.py Oracle tables but native DATE
    in servers and nodes. Mixed representation of "when" across the same ATP.
    Sorting / range purges rely on lexicographic ISO strings on one side and
    real dates on the other.

  * ip_address 255 vs 45, name 100/255/500 : pure width drift. The SMALLER one
    is the risk — a value that fits the wide column would be truncated if ever
    written to the narrow one.

  * uid joins across stores (users."uid" -> bots.uid, reviews.uid) are logical
    only — enforced by application code, not a constraint, because the two
    stores cannot share one.

 No two tables define the SAME column twice within one table (no real duplicate
 within a single CREATE). All "duplicates" are cross-table name reuse.
